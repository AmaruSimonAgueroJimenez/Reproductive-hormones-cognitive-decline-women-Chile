```{python}
#| label: prepare-shared-evidence

from pathlib import Path
import contextlib
import html
import io
import json
import sys
import pandas as pd
from IPython.display import Markdown, HTML, display

# Separate rich-output objects explicitly so Pandoc preserves paragraphs,
# headings and Markdown tables across consecutive Jupyter display calls.
def markdown(text):
    return Markdown("\n\n" + str(text) + "\n\n")

def raw_html(text):
    return HTML("\n\n" + str(text) + "\n\n")

PROJECT = next(p for p in [Path.cwd(), *Path.cwd().parents]
               if (p / "scripts/descriptive_review.py").exists())
sys.path.insert(0, str(PROJECT / "scripts"))
import descriptive_review as review
from descriptive_candidate_bank import load_bank, BANK_BUNDLE

# Recalculate reported numbers from the aggregate tables, fill the manuscript
# templates and construct exactly the same tables and captions as the Word file.
with contextlib.redirect_stdout(io.StringIO()):
    core = review.build_bundle(rebuild_metrics=True)
article = core["languages"]["en"]
core_id = core["content_sha256"]
bank = load_bank() if BANK_BUNDLE.exists() else None
if bank and bank["core_bundle_sha256"] != core_id:
    raise ValueError("Rebuild the candidate bank against the current core evidence")
display(raw_html(f'<div class="evidence-id" data-evidence-bundle="{core_id}">'
             f'Shared core evidence identifier: <code>{core_id}</code></div>'))

def render_blocks(section):
    for block in section["blocks"]:
        prefix = "### " if block["type"] == "heading" else ""
        display(markdown(prefix + block["text"]))

def render_table(table, candidate=False):
    frame = pd.DataFrame(table["rows"], columns=table["columns"])
    title = f'Table {table["number"]} {table["title"]}'
    if candidate:
        display(raw_html('<details class="provenance"><summary>' + html.escape(title)
                     + '</summary>' + frame.to_html(index=False, escape=True)
                     + '<p>' + html.escape(table["note"]) + '</p></details>'))
    else:
        display(markdown("### " + title))
        display(markdown(frame.to_markdown(index=False)))
        display(markdown(table["note"]))

def render_figure(figure):
    raster = PROJECT / figure["path"]
    relative = "../" + str(raster.relative_to(PROJECT))
    display(markdown(f'### Figure {figure["number"]}'))
    display(raw_html(f'<img class="paper-figure" src="{html.escape(relative, quote=True)}" '
                     f'alt="{html.escape(figure["alt"], quote=True)}">'))
    display(markdown(figure["caption"]))
    svg = "../" + str(raster.with_suffix(".svg").relative_to(PROJECT))
    pdf = "../" + str(raster.with_suffix(".pdf").relative_to(PROJECT))
    links = f'[Editable SVG]({svg}) · [Vector PDF]({pdf})'
    if figure.get("prevalence_source_csv"):
        source_table = PROJECT / figure["prevalence_source_csv"]
        if not source_table.is_file():
            raise FileNotFoundError(source_table)
        links += ' · [Plotted comorbidity counts and percentages](../' + figure["prevalence_source_csv"] + ')'
    display(markdown(links))
```

### Reproducing the instrument analyses

The instrument figures, table cells and scientific Methods and Results text are
shared with the Word reports. This page verifies the saved evidence files and
recalculates the explicitly declared aggregate checks at render time. Source
record extraction and original analytical processing are separate operations;
they are not silently repeated when a reader renders this page. The aggregate
inputs and analysis code allow the instrument outputs to be reconstructed
without the external data volume. Original-source refresh additionally requires
the same authorised REM releases and annual coded templates.

The commands below are displayed for inspection and are not executed during
rendering. Use the scientific and document environments recorded by the runner.

## Reproducibility and data access

Aggregate-input reconstruction uses saved study tables, text templates and the
public geographic boundaries packaged with the figures. Source-record
reconstruction requires authorised access to DEIS, GRD, REM, FONASA and INE
files; patient-level records remain outside the repository. GRD encrypted
identifiers are not distributed in these materials.

Rendering this page recalculates the reported core numbers from aggregates.
Network estimation is a separate, explicitly invoked computation with recorded
seeds, replicates and definitions. It is not repeated during every render.
Word consistency checks compare every prose block, every table cell and the
hashes of the embedded figure images.

analysis dependencies are in `requirements.txt`; the versions actually used are
